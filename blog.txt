            // 追踪资源加载进度
            function updateProgress() {
              if (typeof PerformanceObserver !== 'undefined') {
                var resources = performance.getEntriesByType('resource');
                var completed = 0;
                var total = 0;
                
                resources.forEach(function(res) {
                  total++;
                  if (res.responseEnd > 0) completed++;
                });
                
                if (total > 0) {
                  progress = Math.min(92, Math.floor((completed / total) * 100));
                }
              } else {
                // 降级：使用模拟进度
                progress = Math.min(92, progress + Math.max(1, (92 - progress) * 0.1));
              }
              
              if (progressBar) progressBar.style.width = progress + '%';
              if (progressNumber) progressNumber.textContent = Math.round(progress) + '%';
            }
            
            // 定期更新进度
            var progressInterval = setInterval(updateProgress, 100);
            var loaded = false;
            var capTimer = null;

            // 页面加载完成（幂等：只执行一次）
            function handleLoad() {
              if (loaded) return;
              loaded = true;
              clearInterval(progressInterval);
              if (capTimer) clearTimeout(capTimer);
              progress = 100;
              if (progressBar) progressBar.style.width = '100%';
              if (progressNumber) progressNumber.textContent = '100%';

              // 传递进度给 React 组件
              if (typeof window !== 'undefined') {
                window.__SSR_LOADER_PROGRESS__ = 100;
                window.__SSR_LOADER_READY__ = true;
              }

              // Hide the loader but DO NOT removeChild it — that breaks React's
              // reconciliation (NotFoundError: removeChild / insertBefore).
              // Leave it in the DOM as display:none after the fade completes.
              setTimeout(function() {
                if (ssrLoader) {
                  ssrLoader.classList.add('ssr-hidden');
                  setTimeout(function() {
                    if (ssrLoader) {
                      ssrLoader.classList.add('ssr-gone');
